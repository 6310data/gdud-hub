/** עלייה: אם יש snapshot שמור במכשיר מציגים אותו מיד ולא פונים לשרת בכלל. */

function beginBoot(){

  var cached = snapLoad();

  if (cached){ CACHED_BOOT = true; PENDING.snap = cached; maybeStart(); return; }

  loadSnapshot();

}


if (!BOOT.page && window.google && google.script.url && google.script.url.getLocation) {

  google.script.url.getLocation(function(loc){

    var q = (loc && loc.parameter) || {};

    BOOT.arg  = q.c || '';

    if (q.k) TOKEN = q.k;

    ROUTE = { page: BOOT.page, arg: BOOT.arg };

    PENDING.loc = true;

} else {

  if (!BOOT.page) { BOOT.page = 'home'; ROUTE = { page:'home', arg:'' }; }

  beginBoot();

}


function maybeStart(){

  if (PENDING.snap && PENDING.loc) booted(PENDING.snap);

}

function b64download(b64, name, mime){

  var bin = atob(b64), n = bin.length, arr = new Uint8Array(n);

  for (var i=0;i<n;i++) arr[i] = bin.charCodeAt(i);

  var blob = new Blob([arr], { type: mime || 'application/pdf' });

  var u = URL.createObjectURL(blob);

  try {

    var a = document.createElement('a');

    a.href = u; a.download = name;

    document.body.appendChild(a); a.click();

    setTimeout(function(){ a.remove(); }, 1000);

  } catch(e){}

  return u;

}

  function validate(items){

        errs.push('הצ׳ ' + it.newSer + ' כבר קיים במערכת.');

      if (seen[it.newSer]) errs.push('הצ׳ ' + it.newSer + ' מופיע פעמיים בשמירה.');

      seen[it.newSer] = true;

    });

    return errs;

  }

          (res.created ? ' · ' + res.created + ' נוספו כרשומה חדשה' : '') };

        S.open = '';

        refreshSnapshot(function(){ go('serials',''); });

      } else {

        onFail();

        serErr((res && res.errors) ? res.errors : ['שגיאה']);

      }

    }).withFailureHandler(function(err){

      onFail(); serErr([err.message]);

    }).wa_serialSave(TOKEN, { items: items });

  }

function recalcKit(counts){

  var auto = {};

  Object.keys(COMP).forEach(function(k){

    if (COMP[k].manual) counts[k] = (counts[k]||0) + COMP[k].qty;

  });

  Object.keys(counts).forEach(function(d){

    var kit = SNAP.kits[d];

    if(!kit) return;

    Object.keys(kit).forEach(function(item){

      auto[item] = (auto[item]||0) + kit[item]*counts[d];

    });

  });

  var next = {};

  Object.keys(auto).forEach(function(k){

    if (SNAP.stock[k]===undefined) return;

    next[k] = { qty: round2(auto[k]), manual:false };

  });

  Object.keys(COMP).forEach(function(k){

    if (COMP[k].manual) next[k] = COMP[k];

    else if (COMP[k].edited && next[k]){ next[k].qty = COMP[k].qty; next[k].edited = true; }

  });

  Object.keys(COMP).forEach(function(k){ delete COMP[k]; });

  Object.keys(next).forEach(function(k){ COMP[k] = next[k]; });

}

function send(fn, payload, onOk, msgFn){

  $('veil').className='veil on';

  $('go').disabled = true;

  google.script.run

    .withSuccessHandler(function(res){ done(res, onOk, msgFn); })

    .withFailureHandler(function(e){ done({ok:false, errors:[e.message]}, null, msgFn); })

    [fn](TOKEN, payload);

}


function done(res, onOk, msgFn){

  $('veil').className='veil';

  var m = $('msg');

  if (res && res.ok){

    m.className='msg ok';

    m.innerHTML = msgFn ? msgFn(res)

      : ('<b>✅ נשמר</b><br>'+res.serials+' צ׳ ו-'+res.comp+

         ' סוגי ציוד משלים נרשמו על '+esc(res.company)+'.' +

         (res.when ? '<br>נרשם ביומן בתאריך '+esc(res.when)+'.' : ''));

    if (onOk) onOk(res);

    redraw();

  } else {

    m.className='msg err';

    m.innerHTML = '<b>⚠ לא נשמר — שום דבר לא עודכן</b><ul>' +

      (res && res.errors ? res.errors : ['שגיאה לא ידועה'])

        .map(function(x){ return '<li>'+esc(x)+'</li>'; }).join('') + '</ul>';

  }

  window.scrollTo(0,0);

  updateBar();

}
